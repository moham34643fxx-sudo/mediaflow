cmake_minimum_required(VERSION 3.22.1)
project(zmedia C)
set(LAME ${CMAKE_CURRENT_SOURCE_DIR}/vendor/lame-3.100)
if(NOT EXISTS ${LAME}/include/lame.h)
    message(FATAL_ERROR "Run python tools/fetch_native_deps.py first")
endif()
file(GLOB LAME_SOURCES ${LAME}/libmp3lame/*.c)
add_library(mp3lame SHARED ${LAME_SOURCES})
target_include_directories(mp3lame PRIVATE ${LAME}/include ${LAME}/libmp3lame ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(mp3lame PRIVATE HAVE_CONFIG_H)
target_link_libraries(mp3lame m)
target_link_options(mp3lame PRIVATE -Wl,-z,max-page-size=16384)
# Measured on arm64 against O2 with byte-identical CBR/VBR output. Preserve
# floating-point semantics: no fast-math, approximate log, or quality changes.
# Keep ordinary debug builds debuggable; isolated performance reviews opt in.
option(ZMEDIA_RELEASE_ENCODER_REVIEW "Use production encoder optimization in isolated debug reviews" OFF)
if(ANDROID_ABI STREQUAL "arm64-v8a" AND CMAKE_C_COMPILER_ID STREQUAL "Clang")
    if(NOT CMAKE_BUILD_TYPE STREQUAL "Debug" OR ZMEDIA_RELEASE_ENCODER_REVIEW)
        target_compile_options(mp3lame PRIVATE -O3 -flto=thin)
        target_link_options(mp3lame PRIVATE -flto=thin)
    endif()
endif()
add_library(zlame SHARED lame_bridge.c)
target_include_directories(zlame PRIVATE ${LAME}/include)
target_link_libraries(zlame mp3lame)
target_link_options(zlame PRIVATE -Wl,-z,max-page-size=16384)
